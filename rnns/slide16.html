📐 Equações e Ativações

fₜ = σ(Wf · [hₜ₋₁, xₜ] + bf)   ≈ 0.62
iₜ = σ(Wi · [hₜ₋₁, xₜ] + bi)   ≈ 0.73
c̃ₜ = tanh(Wc · [hₜ₋₁, xₜ] + bc)  ≈ 0.46
cₜ = fₜ ⊙ cₜ₋₁ + iₜ ⊙ c̃ₜ      ≈ 0.83
oₜ = σ(Wo · [hₜ₋₁, xₜ] + bo)   ≈ 0.68
hₜ = oₜ ⊙ tanh(cₜ)          ≈ 0.48
